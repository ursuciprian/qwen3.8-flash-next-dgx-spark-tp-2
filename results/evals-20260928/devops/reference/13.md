groups:
  - name: app
    rules:
      - record: service:http_requests_5xx:ratio_rate5m
        expr: sum by (service) (rate(http_requests_total{status=~"5.."}[5m])) / sum by (service) (rate(http_requests_total[5m]))
      - alert: HighErrorRate
        expr: service:http_requests_5xx:ratio_rate5m > 0.05
        for: 10m
        labels:
          severity: page
        annotations:
          summary: "{{ $labels.service }} 5xx ratio is {{ $value | humanizePercentage }}"
      - alert: PodCrashLooping
        expr: increase(kube_pod_container_status_restarts_total[15m]) > 3
        labels:
          severity: ticket
        annotations:
          summary: "{{ $labels.namespace }}/{{ $labels.pod }} restarting"
