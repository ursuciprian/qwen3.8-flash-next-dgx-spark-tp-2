```bash
#!/usr/bin/env bash
set -euo pipefail

usage() {
  cat <<USAGE
Usage: pg-backup.sh --db NAME [--out-dir DIR] [--keep N] [--dry-run]
USAGE
}

log() { printf '%s %s\n' "$(date '+%Y-%m-%dT%H:%M:%S')" "$*" >&2; }

db="" out_dir="/var/backups/pg" keep=7 dry_run=0
while [[ $# -gt 0 ]]; do
  case "$1" in
    --db) db="${2:-}"; shift 2 ;;
    --out-dir) out_dir="${2:-}"; shift 2 ;;
    --keep) keep="${2:-}"; shift 2 ;;
    --dry-run) dry_run=1; shift ;;
    -h|--help) usage; exit 0 ;;
    *) usage >&2; exit 2 ;;
  esac
done
[[ -n "$db" ]] || { usage >&2; exit 2; }
[[ "$keep" =~ ^[0-9]+$ ]] || { usage >&2; exit 2; }

file="$out_dir/$db-$(date '+%Y%m%dT%H%M%S').sql.gz"
if (( dry_run )); then
  log "dry-run: would dump $db to $file and keep $keep backups"
  exit 0
fi
mkdir -p "$out_dir"
tmp="$file.partial"
trap 'rm -f "$tmp"' EXIT
log "dumping $db to $file"
pg_dump "$db" | gzip > "$tmp"
mv "$tmp" "$file"
mapfile -t old < <(find "$out_dir" -maxdepth 1 -name "$db-*.sql.gz" | sort -r | tail -n +"$((keep + 1))")
for f in "${old[@]}"; do
  log "removing $f"
  rm -f -- "$f"
done
```
